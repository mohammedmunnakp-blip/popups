<script>
const BLOCK_ID = "PUT_YOUR_ADSGRAM_BLOCK_ID_HERE";   // your AdsGram block ID
const SHARE_URL = "https://t.me/YOUR_BOT/YOUR_APP";  // your Mini App direct link
const CAP = 4;
const COLORS = ["#ef4444","#f59e0b","#facc15","#22c55e","#06b6d4","#3b82f6","#8b5cf6","#ec4899","#f97316","#84cc16"];
const $ = id => document.getElementById(id);
const store = {
  get(k, d) { try { const v = localStorage.getItem(k); return v === null ? d : JSON.parse(v); } catch (e) { return d; } },
  set(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); } catch (e) {} }
};
let level = store.get("cp_level", 1), maxLevel = store.get("cp_max", 1), stars = store.get("cp_stars", {});
let daily = store.get("cp_daily", { last: "", streak: 0, done: {} }), sound = store.get("cp_sound", true);
let mode = "level", tubes = [], start = "", sel = -1, hintB = -1, history = [], moves = 0, extraUsed = false, hints = 3, par = 12, lastStars = 0;
let ctx = null;

try { Telegram.WebApp.ready(); Telegram.WebApp.expand(); } catch (e) {}

const today = () => new Date().toISOString().slice(0, 10);
const yesterday = () => new Date(Date.now() - 864e5).toISOString().slice(0, 10);
function rng(seed) { return function () { seed |= 0; seed = seed + 0x6D2B79F5 | 0; let t = Math.imul(seed ^ seed >>> 15, 1 | seed); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; }; }
function shuffle(a, r) { for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(r() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; } }
const solvedTube = t => t.length === CAP && t.every(x => x === t[0]);
function toast(m) { const t = $("toast"); t.textContent = m; t.style.opacity = 1; clearTimeout(toast.h); toast.h = setTimeout(() => t.style.opacity = 0, 2400); }
function buzz() { try { Telegram.WebApp.HapticFeedback.impactOccurred("light"); } catch (e) {} }
function beep(f, d = .08) {
  if (!sound) return;
  try {
    ctx = ctx || new (window.AudioContext || window.webkitAudioContext)();
    const o = ctx.createOscillator(), g = ctx.createGain();
    o.frequency.value = f; g.gain.value = .05; o.connect(g); g.connect(ctx.destination);
    o.start(); o.stop(ctx.currentTime + d);
  } catch (e) {}
}

function newLevel() {
  let n, r;
  if (mode === "daily") { n = 6; r = rng(parseInt(today().replace(/-/g, ""))); }
  else { n = Math.min(3 + Math.floor((level - 1) / 2), COLORS.length); r = Math.random; }
  par = n * 3;
  const pool = [];
  for (let c = 0; c < n; c++) for (let i = 0; i < CAP; i++) pool.push(c);
  do {
    shuffle(pool, r);
    tubes = [];
    for (let t = 0; t < n; t++) tubes.push(pool.slice(t * CAP, t * CAP + CAP));
    tubes.push([], []);
  } while (tubes.some(solvedTube));
  start = JSON.stringify(tubes);
  sel = -1; hintB = -1; history = []; moves = 0; extraUsed = false; hints = 3;
  $("win").style.display = "none";
  render();
}

function render() {
  $("modeLbl").textContent = mode === "daily" ? "📅 Daily" : "Level " + level;
  $("mv").textContent = moves;
  $("streak").textContent = "🔥 " + (daily.last === today() || daily.last === yesterday() ? daily.streak : 0);
  $("hint").textContent = "💡 Hint (" + hints + ")";
  $("snd").textContent = sound ? "🔊" : "🔇";
  const b = $("board"); b.innerHTML = "";
  tubes.forEach((t, i) => {
    const d = document.createElement("div");
    d.className = "tube" + (i === sel ? " sel" : "") + (i === hintB ? " hint" : "");
    t.forEach(c => { const s = document.createElement("div"); s.className = "seg"; s.style.background = COLORS[c]; d.appendChild(s); });
    d.onclick = () => tap(i);
    b.appendChild(d);
  });
}

const canPour = (a, b) => {
  const A = tubes[a], B = tubes[b];
  return a !== b && A.length && B.length < CAP && (!B.length || B[B.length - 1] === A[A.length - 1]);
};
function pour(a, b) {
  if (!canPour(a, b)) return false;
  const A = tubes[a], B = tubes[b], c = A[A.length - 1];
  history.push(JSON.stringify(tubes));
  while (A.length && A[A.length - 1] === c && B.length < CAP) B.push(A.pop());
  moves++;
  return true;
}
function findHint() {
  for (let pass = 0; pass < 2; pass++)
    for (let a = 0; a < tubes.length; a++) {
      if (solvedTube(tubes[a]) || !tubes[a].length) continue;
      for (let b = 0; b < tubes.length; b++) {
        if (!canPour(a, b)) continue;
        const toEmpty = tubes[b].length === 0;
        if (pass === 0 && !toEmpty) return [a, b];
        if (pass === 1 && toEmpty && !tubes[a].every(x => x === tubes[a][0])) return [a, b];
      }
    }
  return null;
}

function tap(i) {
  hintB = -1;
  if (sel < 0) { if (tubes[i].length) { sel = i; buzz(); beep(500); } }
  else if (sel === i) sel = -1;
  else if (pour(sel, i)) {
    sel = -1; buzz(); beep(700);
    if (!checkWin() && !findHint()) toast("No useful moves left. Try Undo, Restart or an Extra tube.");
  } else sel = tubes[i].length ? i : -1;
  render();
}

function checkWin() {
  if (!tubes.every(t => t.length === 0 || solvedTube(t))) return false;
  lastStars = moves <= par ? 3 : moves <= par * 1.5 ? 2 : 1;
  if (mode === "daily") {
    if (!daily.done[today()]) {
      daily.streak = daily.last === yesterday() ? daily.streak + 1 : 1;
      daily.last = today(); daily.done[today()] = lastStars; store.set("cp_daily", daily);
    }
  } else {
    stars[level] = Math.max(stars[level] || 0, lastStars); store.set("cp_stars", stars);
    if (level >= maxLevel) { maxLevel = level + 1; store.set("cp_max", maxLevel); }
    store.set("cp_level", Math.min(level + 1, maxLevel));
  }
  $("winmsg").textContent = "⭐".repeat(lastStars) + "  " + moves + " moves";
  $("next").textContent = mode === "daily" ? "Back to levels ▶" : "Next level ▶";
  $("win").style.display = "flex";
  beep(880, .25);
  return true;
}

async function watchAd() {
  if (!window.Adsgram || BLOCK_ID.startsWith("PUT_")) { toast("Ads are available inside Telegram only."); return false; }
  try { await window.Adsgram.init({ blockId: BLOCK_ID }).show(); return true; }
  catch (e) { toast("No ad available right now. Try again later."); return false; }
}
function share() {
  const text = "I solved " + (mode === "daily" ? "today's daily" : "level " + level) + " in ColorPour with " + lastStars + " stars! Can you beat me?";
  const url = "https://t.me/share/url?url=" + encodeURIComponent(SHARE_URL) + "&text=" + encodeURIComponent(text);
  try { Telegram.WebApp.openTelegramLink(url); } catch (e) { window.open(url, "_blank"); }
}
function openLevels() {
  const g = $("grid"); g.innerHTML = "";
  const total = Math.max(maxLevel + 4, 20);
  for (let i = 1; i <= total; i++) {
    const b = document.createElement("button");
    b.className = i > maxLevel ? "locked" : "";
    b.innerHTML = i + "<small>" + (i <= maxLevel ? "⭐".repeat(stars[i] || 0) : "🔒") + "</small>";
    b.onclick = () => { level = i; mode = "level"; $("lvov").style.display = "none"; newLevel(); };
    g.appendChild(b);
  }
  $("lvov").style.display = "flex";
}

$("undo").onclick = () => { if (history.length) { tubes = JSON.parse(history.pop()); moves = Math.max(0, moves - 1); sel = hintB = -1; render(); } };
$("restart").onclick = () => { tubes = JSON.parse(start); sel = hintB = -1; history = []; moves = 0; extraUsed = false; render(); };
$("hint").onclick = () => {
  if (hints <= 0) return toast("No hints left on this level.");
  const h = findHint();
  if (!h) return toast("No useful move found. Try Undo or Restart.");
  hints--; sel = h[0]; hintB = h[1]; render();
};
$("extra").onclick = async () => {
  if (extraUsed) return toast("Extra tube already used on this level.");
  if (await watchAd()) { tubes.push([]); extraUsed = true; render(); toast("Extra tube added!"); }
};
$("skip").onclick = async () => {
  if (mode === "daily") return toast("The daily challenge can't be skipped.");
  if (await watchAd()) { level++; maxLevel = Math.max(maxLevel, level); store.set("cp_max", maxLevel); store.set("cp_level", level); newLevel(); }
};
$("daily").onclick = () => {
  if (daily.done[today()]) return toast("Daily done! Come back tomorrow. 🔥 " + daily.streak + " day streak");
  mode = "daily"; newLevel();
};
$("snd").onclick = () => { sound = !sound; store.set("cp_sound", sound); render(); };
$("levels").onclick = openLevels;
$("lvclose").onclick = () => $("lvov").style.display = "none";
$("share").onclick = share;
$("next").onclick = () => { if (mode === "daily") { mode = "level"; level = Math.min(store.get("cp_level", 1), maxLevel); } else level++; newLevel(); };

level = Math.min(level, maxLevel);
newLevel();
</script>
